<?php

namespace App\Http\Requests;

use App\Models\ResearchSchema;
use Illuminate\Foundation\Http\FormRequest;

class StoreSchemaRequest extends FormRequest
{
    public function authorize(): bool
    {
        return $this->user()->can('create', ResearchSchema::class);
    }

    public function rules(): array
    {
        return [
            'nama_skema' => ['required', 'string', 'max:255'],
            'deskripsi' => ['nullable', 'string', 'max:5000'],
            'biaya_maksimal' => ['required', 'numeric', 'min:1000000'],
            'tanggal_mulai' => ['required', 'date'],
            'tanggal_selesai' => ['required', 'date', 'after_or_equal:tanggal_mulai'],
            'file_panduan' => ['nullable', 'file', 'mimes:pdf', 'max:10240'],
            'is_active' => ['boolean'],
            // Aliases support
            'name' => ['nullable', 'string', 'max:255'],
            'description' => ['nullable', 'string', 'max:5000'],
            'max_funding' => ['nullable', 'numeric'],
            'start_date' => ['nullable', 'date'],
            'end_date' => ['nullable', 'date'],
        ];
    }

    protected function prepareForValidation(): void
    {
        if (! $this->has('nama_skema') && $this->filled('name')) {
            $this->merge(['nama_skema' => $this->input('name')]);
        }
        if (! $this->has('deskripsi') && $this->filled('description')) {
            $this->merge(['deskripsi' => $this->input('description')]);
        }
        if (! $this->has('biaya_maksimal') && $this->filled('max_funding')) {
            $this->merge(['biaya_maksimal' => $this->input('max_funding')]);
        }
        if (! $this->has('tanggal_mulai') && $this->filled('start_date')) {
            $this->merge(['tanggal_mulai' => $this->input('start_date')]);
        }
        if (! $this->has('tanggal_selesai') && $this->filled('end_date')) {
            $this->merge(['tanggal_selesai' => $this->input('end_date')]);
        }
        if (! $this->has('tanggal_mulai')) {
            $this->merge(['tanggal_mulai' => now()->toDateString()]);
        }
        if (! $this->has('tanggal_selesai')) {
            $this->merge(['tanggal_selesai' => now()->addMonths(3)->toDateString()]);
        }
        if (! $this->has('biaya_maksimal')) {
            $this->merge(['biaya_maksimal' => 50000000]);
        }
    }
}
